1. Infidelity rate by marital satisfaction

affairs %>%
  group_by(ratemarr) %>%
  summarise(affair_rate = mean(affair) * 100) %>%
  ggplot(aes(x = factor(ratemarr), y = affair_rate)) +
  geom_col() +
  labs(
    title = "Infidelity Rate by Marital Satisfaction",
    x = "Marital Satisfaction Rating",
    y = "Affair Rate (%)"
  ) +
  theme_minimal()

ggsave("marital_satisfaction_affair_rate.png", width = 8, height = 5)

4. Average marginal effects chart

marginal_effects <- data.frame(
  variable = c("Age", "Education", "Kids", "Marital Satisfaction 2",
               "Marital Satisfaction 3", "Marital Satisfaction 4",
               "Marital Satisfaction 5", "Religiosity", "Years Married"),
  ame = c(-0.0065, 0.0089, 0.0603, -0.0019,
          -0.1775, -0.2342, -0.3078, -0.0536, 0.0149)
)

ggplot(marginal_effects, aes(x = reorder(variable, ame), y = ame)) +
  geom_col() +
  coord_flip() +
  labs(
    title = "Average Marginal Effects from Logit Model",
    x = "Variable",
    y = "Average Marginal Effect"
  ) +
  theme_minimal()

ggsave("marginal_effects_chart.png", width = 8, height = 5)
